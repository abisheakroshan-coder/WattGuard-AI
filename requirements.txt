    data = pd.read_csv(OUTPUT_FILE)

    return data

def metric_card(
    number,
    label,
    style=""
):

    return f"""
    <div class="kpi-card {style}">
        <div class="kpi-number">
            {number}
        </div>

        <div class="kpi-label">
            {label}

def risk_badge(level):

    level = str(level).upper()

    if level == "CRITICAL":

        return """
        <span class="risk-critical">
        CRITICAL
        </span>
        """

    elif level == "HIGH":

        return """
        <span class="risk-high">
        HIGH
        </span>
        """

    else:

        return f"""
        <span class="risk-low">
        {level}
        </span>
        """


def consumer_index_from_id(
    consumer_id
):

    try:

        return int(
            str(consumer_id)
            .replace("WG-", "")
        )

    except:

        return None


def get_history(
    consumer_id
):

    if raw_df is None:

        return None

    idx = consumer_index_from_id(
        consumer_id
    )

    if idx is None:

        return None

    if idx not in raw_df.index:

        return None


    row = raw_df.loc[idx]


    history = []


    for column in raw_df.columns:

        date_value = pd.to_datetime(
            column,
            errors="coerce"
        )

        if pd.isna(date_value):

            continue


        value = pd.to_numeric(
            row[column],
            errors="coerce"
        )


        if pd.isna(value):

            continue


        history.append(
            {
                "Date": date_value,
                "Usage": float(value)
            }
        )


    if not history:

        return None


    result = pd.DataFrame(history)

    result = result.sort_values(
        "Date"
    )


    result["30-Day Average"] = (

        result["Usage"]
        .rolling(
            window=30,
            min_periods=1
        )
        .mean()

    )


    return result


def create_field_note(row):

    reasons = str(
        row["Why_Flagged"]
    ).replace(
        "|",
        ","
    )


    return (

        f"{row['Consumer_ID']} was prioritised because "
        f"its consumption behaviour differs from the normal pattern. "
        f"Key indicators: {reasons}. "
        f"The model estimates approximately "
        f"{row['Estimated_Unbilled_Units']:.1f} units of possible "
        f"unexplained consumption. "
        f"Recommended action: {row['Action'].lower()}."
    )
